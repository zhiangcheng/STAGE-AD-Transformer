cd /media/desk15/iy2120/Project2026/Myproject_AD/data/

1. Download the 1000G Phase 3 VCF files and sample panel
mkdir -p 1000G_Phase3_v5
cd 1000G_Phase3_v5

base=https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502

# sample panel
wget -c ${base}/integrated_call_samples_v3.20130502.ALL.panel

# chromosomes 1-22
for chr in {1..22}; do
  wget -c ${base}/ALL.chr${chr}.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz
  wget -c ${base}/ALL.chr${chr}.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz.tbi
done

2. Generate the keep file for EUR samples
awk 'NR > 1 && $3 == "EUR" {print $1, $1}' \
  integrated_call_samples_v3.20130502.ALL.panel > EUR_population_IDs.txt

3. Convert to a PLINK reference panel
source activate GWAS
for chr in {1..22}; do
  plink2 \
    --vcf ALL.chr${chr}.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz \
    --double-id \
    --keep EUR_population_IDs.txt \
    --max-alleles 2 \
    --snps-only just-acgt \
    --set-all-var-ids @:#:\$r:\$a \
    --new-id-max-allele-len 100 missing \
    --make-bed \
    --out ALL.chr${chr}.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes_CHRBPREFALT_EUR
done

4. Remove duplicates and generate the final clumping panel
for chr in {1..22}; do
  plink2 \
    --bfile ALL.chr${chr}.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes_CHRBPREFALT_EUR \
    --rm-dup exclude-all \
    --make-bed \
    --out ALL.chr${chr}.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes_CHRBPA1A2_no_missing_dups
done
